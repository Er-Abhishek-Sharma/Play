Network NTP Clock Estimation

SyncWave uses Cristian's / NTP algorithm over WebSocket pings. Each device performs 8 high-resolution probe bursts, discards outliers, and computes the relative offset $\theta$ and Round-Trip Time (RTT).

RTT = (T4 - T1) - (T3 - T2)
Clock Offset = ((T2 - T1) + (T3 - T4)) / 2

import uuid
from django.db import models

class Room(models.Model):
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    code = models.CharField(max_length=12, unique=True, db_index=True)
    name = models.CharField(max_length=100)
    host = models.ForeignKey(User, on_delete=models.CASCADE, related_name='hosted_rooms')
    is_spatial_enabled = models.BooleanField(default=True)
    is_collaborative_queue = models.BooleanField(default=True)
    created_at = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return f"Room {self.code} ({self.name})"

class Device(models.Model):
    DEVICE_TYPES = [
        ('laptop', 'Laptop'),
        ('phone', 'Smartphone'),
        ('tablet', 'Tablet'),
        ('desktop', 'Desktop'),
        ('tv', 'Smart TV'),
    ]
    room = models.ForeignKey(Room, on_delete=models.CASCADE, related_name='devices')
    session_id = models.CharField(max_length=64, unique=True)
    name = models.CharField(max_length=100)
    device_type = models.CharField(max_length=20, choices=DEVICE_TYPES, default='phone')
    clock_offset_ms = models.FloatField(default=0.0)
    hardware_latency_ms = models.FloatField(default=25.0)
    volume = models.FloatField(default=0.85)
    is_muted = models.BooleanField(default=False)
    stage_x = models.FloatField(default=50.0)
    stage_y = models.FloatField(default=50.0)
    last_ping = models.DateTimeField(auto_now=True)

# syncwave/synchronization/consumers.py
import json
import time
from channels.generic.websocket import AsyncWebsocketConsumer

class RoomConsumer(AsyncWebsocketConsumer):
    async def connect(self):
        self.room_code = self.scope['url_route']['kwargs']['room_code']
        self.room_group_name = f'syncwave_room_{self.room_code}'

        # Join Redis Channel Layer Group
        await self.channel_layer.group_add(
            self.room_group_name,
            self.channel_name
        )
        await self.accept()

    async def disconnect(self, close_code):
        await self.channel_layer.group_discard(
            self.room_group_name,
            self.channel_name
        )

    async def receive(self, text_data):
        payload = json.loads(text_data)
        event_type = payload.get('type')
        client_time = payload.get('client_time', 0.0)
        server_now = time.time() * 1000.0  # Precise milliseconds

        # High-precision NTP Ping/Pong Handling
        if event_type == 'PING':
            await self.send(text_data=json.dumps({
                'type': 'PONG',
                'client_t1': client_time,
                'server_t2': server_now,
                'server_t3': time.time() * 1000.0
            }))
            return

        # Synchronized Playback Scheduling Broadcast
        if event_type == 'SCHEDULE_PLAY':
            # Lead time buffer ensures all devices pre-buffer and trigger on exact millisecond
            schedule_time = server_now + 1200.0  
            await self.channel_layer.group_send(
                self.room_group_name,
                {
                    'type': 'broadcast_playback',
                    'action': 'PLAY',
                    'position': payload.get('position', 0.0),
                    'scheduled_start': schedule_time,
                    'track_id': payload.get('track_id')
                }
            )

    async def broadcast_playback(self, event):
        await self.send(text_data=json.dumps(event))

# syncwave/synchronization/sync_engine.py
"""
High-Precision Cristian's NTP Clock Synchronization Algorithm
Used by SyncWave to establish a unified virtual room clock.
"""
import numpy as np

def compute_ntp_offset(samples):
    """
    samples is a list of tuples: (t1, t2, t3, t4)
    t1: client send time
    t2: server receive time
    t3: server response send time
    t4: client receive time
    """
    offsets = []
    rtts = []
    for t1, t2, t3, t4 in samples:
        rtt = (t4 - t1) - (t3 - t2)
        offset = ((t2 - t1) + (t3 - t4)) / 2.0
        rtts.append(rtt)
        offsets.append(offset)

    # Filter out top 25% highest RTT outliers (network jitter protection)
    sorted_indices = np.argsort(rtts)
    best_samples = sorted_indices[:max(1, int(len(samples) * 0.75))]
    calibrated_offset = np.median([offsets[i] for i in best_samples])
    return calibrated_offset
